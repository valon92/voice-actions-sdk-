🌐 Hosted SDK - User Settings Example

Demonstrim se si funksionon user-level settings me hosted SDK nga CDN

✅ Po, funksionon e njëjta!
Hosted SDK nga voiceactions.dev ka të njëjtin funksionalitet si npm package, përfshirë user-level settings, widget, wake words, notifications, dhe të gjitha features të tjera.

📝 Shembull i Plotë

<!-- Load SDK direkt nga voiceactions.dev -->
<script src="https://voiceactions.dev/sdk/voice-actions-sdk.min.js"></script>
<script>
  // Configuration
  const API_KEY = 'va_your-api-key-here';
  const PLATFORM_NAME = 'your-platform-name';
  const USER_ID = 'user123'; // Current user ID
  const API_URL = 'https://api.voiceactions.dev/api';

  let sdk = null;
  let widget = null;

  // Initialize SDK me userIdentifier
  sdk = new VoiceActionsSDK({
    apiKey: API_KEY,
    platform: PLATFORM_NAME,
    locale: 'en-US',
    userIdentifier: USER_ID, // ⚠️ IMPORTANT: Për user-level settings!
    apiUrl: API_URL,
    onCommand: (command) => {
      console.log('🎤 Command:', command);
      // Handle command
    },
    onError: (error) => {
      console.error('Error:', error);
    }
  });

  // Initialize Widget
  // Widget do të shfaqet vetëm nëse user ka enabled Voice Actions
  widget = new VoiceActionsWidget({
    sdk: sdk,
    position: 'bottom-right',
    autoCheck: true, // Auto-check user settings çdo 30 sekonda
    checkInterval: 30000
  });

  // SDK do të kontrollojë automatikisht:
  // 1. Platform-level setting (nëse platforma ka enabled)
  // 2. User-level setting (nëse user ka enabled)
  // 3. Nëse user ka disabled, SDK nuk do të inicializohet
  // 4. Widget nuk do të shfaqet nëse user ka disabled
</script>

🔧 User Settings API

// Check nëse user ka enabled Voice Actions
async function checkUserEnabled() {
  const response = await fetch(
    `${API_URL}/user-voice-settings/check?user_identifier=${USER_ID}`,
    {
      headers: {
        'X-API-Key': API_KEY
      }
    }
  );

  const data = await response.json();
  return data.enabled; // true ose false
}

// Update user settings
async function updateUserSettings(enabled) {
  await fetch(`${API_URL}/user-voice-settings`, {
    method: 'PUT',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': API_KEY
    },
    body: JSON.stringify({
      user_identifier: USER_ID,
      voice_actions_enabled: enabled
    })
  });
}
